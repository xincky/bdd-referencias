 WHERE estado = 'ENVIADA' ORDER BY fecha_envio LIMIT 100

                                Index Cond: ((region_destino = 'CUS'::bpchar) AND (estado = 'ENVIADA'::text))
              ->  Bitmap Heap Scan on referencia
                    Recheck Cond: (estado = 'ENVIADA'::text)
                    ->  Bitmap Index Scan on referencia_estado_idx
                          Index Cond: (estado = 'ENVIADA'::text)
              ->  Bitmap Heap Scan on referencia referencia_1
